
def defs(tree):
 out={}
 for x in tree.body:
  if isinstance(x,(ast.FunctionDef,ast.AsyncFunctionDef)):out[x.name]=x
  if isinstance(x,ast.ClassDef):
   for y in x.body:
    if isinstance(y,ast.FunctionDef):out[x.name+'::'+y.name]=y
 return out
